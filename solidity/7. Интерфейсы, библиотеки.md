```solidity

// Logger.sol
contract Logger {
    mapping(address => uint[]) payments;
    
    function log(address _from, uint _amount) public {
        require(_from != address(0), "0x0 address");
    
        payments[from].push(_amount);
    }
    
    function getEntry(address _from, uint _index) public view returns(uint) {
        return payments[_from][_index];
    }
}

```

Возникает необходимость логгирования. Если Logger уже где-то развернут, то нужно взять его адрес. 

```solidity

import "./Logger.sol";

// Demo.sol
contract Demo {
    Logger logger;
    constructor(address _logger) {
        logger = Logger(_logger);
    }

    function payment(address _from, uint _index) public view returns(uint) {
        return logger.getEntry(_from, _index);
    }

    receive() external payable {
        logger.log(msg.sender, msg.value);
    }
}
```

Но это если есть код контракта.

В интерфейсе прописывается, какие функции доступны (их сигнатуры), реализацию их мы не делаем.

Даже если в контракте функция public, в интерфейсе должен быть external.


```solidity
// ILogger.sol

interface ILogger {
    function log(address _from, uint _amount) external;
    function getEntry(address _from, uint _index) external view returns(uint);
}
```


```solidity

// Logger.sol
contract Logger is ILogger {    // впринципе не обязательно писать is
    mapping(address => uint[]) payments;
    
    function log(address _from, uint _amount) public {
        require(_from != address(0), "0x0 address");
        payments[from].push(_amount);
    }
    
    function getEntry(address _from, uint _index) public view returns(uint) {
        return payments[_from][_index];
    }
}
```

```solidity

import "./ILogger.sol";

// Demo.sol
contract Demo {
  ILogger logger;
  constructor(address _logger) {
    logger = ILogger(_logger);
  }

  function payment(address _from, uint _index) public view returns(uint) {
    return logger.getEntry(_from, _index);
  }

  receive() external payable {
    logger.log(msg.sender, msg.value);
  }
}
```


### Библиотеки

Набор функций, которые можно использовать:

```solidity
// StrExt.sol

library StrExt {
    function eq(string memory a, strin memory b) internal pure returns(bool) {
        return keccak256(abi.encodePacked(a)) == keccak256(abi.encodePacked(b)); 
    }
}

library ArrayExt {
    function inArray(uint[] a, uint value) internal pure returns(bool) {
        for (uint i = 0; i < a.length; i++) {
            if (a[i] == value) {
                return true;
            }
        }

        return false;
    }
}
```

```solidity

import "./StrExt.sol";

// Demo.sol
contract Demo {
    using StrExt for string;
    using ArrayExt for uint[];
    
    function test() public pure {
        string memory a = "a";
        string memory b = "a";
    
        a.eq(b); // чаще делают так
        StrExt.eq(a, b);
    
        uint[] memory numbers = [1,2,3];
        uint number;
    
        numbers.inArray(number);
    }
}
```
