### call

Ситуация: есть контракт, у нас нет кода его контракта, но мы хотим обращаться к нему. Для этого есть низкоуровневые вызовы:

```solidity

// контракт, кода которого у нас нет
contract AnotherContract {
    string public name;
    mapping (address => uint) public balances;

    function setName(string calldata _name) external view returns (bool){
        name = _name;
        return true;
    }

    receive() external payable {
        balances[msg.sender] += msg.value;
    }
}
```

#### Отличие call от transfer:
1. В отличие от call, transfer сам порождает ошибку и откатывает транзакцию.
2. В transfer есть лимит по газу - 2300



```solidity
contract MyContract {
    address otherContract;

    constructor(address _address) {
        otherContract = _address;
    }

    event Response(bytes response);

    function callReceive() external payable {
        (bool success, ) = otherContract.call{value: msg.value}("");
        require(success, "cant send funds");
    }

    function callSetName(string calldata _name) external {
        (bool success, bytes memory response) = otherContract.call{}(
            abi.encodeWithSignature("setName(string)", _name)
        );

        require(success, "cant set name");

        emit Response(response);
    }

    // если есть код контракта
    function callSetName2(string calldata _name) external {
        (bool success, bytes memory response) = otherContract.call{}(
            abi.encodeWithSelector(AnotherContract.setName.selector, _name)
        );

        require(success, "cant set name");

        emit Response(response);
    }

}
```

### delegatecall


```solidity
contract AnotherContract {
    address public sender;
    uint public amount;

    function getData() external payable {
        sender = msg.sender;
        amount = msg.value;
        emit Received(msg.sender, msg.value);
    }
}
```


```solidity
contract MyContract {
    address public sender;
    uint public amount;

    address otherContract;

    constructor(address _address) {
        otherContract = _address;
    }

    function delegateCallGetData(string calldata _name) external payable {
        (bool success, ) = otherContract.delegatecall(
            abi.encodeWithSelector(AnotherContract.getData.selector);
        );

        require(success, "");
    }


}
```

После получения event, в Received будет адрес аккаунта, который вызвал функцию delegateCallGetData, а не адрес MyContract, в отличие от call. 
Это происходит потому, что функция getData выполняется в конктексте MyContract, а не в контексте AnotherContract.

Кроме того, переменные sender, amount из AnotherContract не будут установлены, но в MyContract sender, amount будут установлены значениями msg.sender, msg.value. 

Но важно, чтобы переменные шли в том же порядке и по тем же слотам.

Например:




```solidity
contract AnotherContract {
    address public sender;
    uint public amount;

    function getData() external payable {
        sender = msg.sender;
        amount = msg.value;
        emit Received(msg.sender, msg.value);
    }
}
```


```solidity
contract MyContract {
    address otherContract;
    address owner;
    address public sender;
    uint public amount;

    constructor(address _address) {
        otherContract = _address;
    }

    function delegateCallGetData(string calldata _name) external payable {
        (bool success, ) = otherContract.delegatecall(
            abi.encodeWithSelector(AnotherContract.getData.selector);
        );

        require(success, "");
    }
}
```

В данном случае, otherContract и owner перезапишутся значениями msg.sender, msg.value, что открывает возможность к взлому контракта.
