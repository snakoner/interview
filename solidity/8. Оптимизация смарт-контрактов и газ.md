### Оптимизация смарт-контрактов и газ

1. В solidity не нужно создавать лишние переменные и функции, так как деплой будет дороже.
2. Не нужно для value type state variables присваивать значение по умолчанию, если оно не отлично от 0, так как так и так будет значение 0.
3. Упаповка state variables: если несколько переменных можно поместить в один слот (256 бит), то их нужно объявлять рядом.
4. Меньшая размерность переменной не означает меньше газа. Если переменная меньше 1 слота, но будет занимать 1 слот, то такая операция более дорогая, так как ее нужно будет предварительно обрезать.
5. Если есть возможность задавать статические значения для переменных, то лучше делать так, нежели вычислять их динамически при деплое.
6. Без необходимости не использовать промежуточные переменные, так как работа с ними тоже стоит газа.
7. По возможности лучше использовать mapping, а не динамические массивы.
8. Динамические массивы дороже, чем статические массивы (push требудет газа).
9. Для массивов важно, какая размерность у value type переменной, так как он упаковывает значения.
10. Лучше не создавать много мелких функций
11. В require лучше использовать более короткие message, желательно, чтоб строка умещалась в 32 байта.
12. Меньше менять переменную состояния.
13. Лучше использовать сторонние библиотеки.


```solidity
contract Optimized {
    uint demo; // (2)

    uint128 a = 1;    // (3)
    uint128 b = 1;    // a и b будут в одном слоте, всего занято 2 слота на a,b,c
    uint256 c = 1;    //

    uint256 demo1 = 1;  // (4)

    bytes32 public hash = 0x9c22ff5f21f0b81b113e63f7db6da94fedef11b2119b4088b89664fb9a3cb658; // (5)


    mapping (address => uint) payments;      // (6)
    function pay() external payable {
        require(msg.sender != address(0), "zero address");
        payments[msg.sender] = msg.value;
    }

    uint8[] demo = [1,2,3];    // (9)

    uint public result = 1;    // (12)
    function doWork(uint[] memory data) public {
        uint res = 1;
        for (uint i = 0; i < data.length; i++) {
            res *= data[i];
        }

        result = res;
    }    
}

contract Unoptimized {
    uint demo = 0; // (2)

    uint128 a = 1;    // (3)
    uint256 c = 1;    // 
    uint128 b = 1;    // a и b будут в разных слотах, всего занято 3 слота на a,b,c

    uint8 demo1 = 1;  // (4)

    bytes32 public hash = keccak256(abi.encodedPacked("test"));    // (5)

    mapping (address => uint) payments;    // (6)
    function pay() external payable {
        address _from = msg.sender;
        require(_from != address(0), "zero address");
        payments[_from] = msg.value;
    }

    uint[] demo = [1,2,3];    // (9)

    uint public result = 1;    // (12)
    function doWork(uint[] memory data) public {
        for (uint i = 0; i < data.length; i++) {
            result *= data[i];
        }
    }    

}
```
