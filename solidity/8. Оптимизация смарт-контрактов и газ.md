1. ++value стоит меньше, чем value++
2. Дешевле использовать mapping напрямую, нежели через access-функцию:
```solidity
mapping (address => uint256) private _balances;

// non optimized
function balanceOf(address account) internal view returns (uint256) {
    return _balances[account];
}

address account = address(this);
uint256 balance = balanceOf(account);

// optimized
address account = address(this);
uint256 balance = _balances[account];

```

3. Более дешевые проверки входных данных(require) должны предшествовать дорогостоящим операциям.
4. Сохранение ссылки на переменную состояние вместо ее повторного извлечения:
```solidity

// non optimized
if (checkpoints[nftId][nCheckpoints - 1].fromBlock <= blockNumber) {
    return checkpoints[nftId][nCheckpoints - 1].delegatedTokenIds;
}

// optimized
Checkpoint storage checkpoint = checkpoints[nftId][nCheckpoints - 1];
if (checkpoint.fromBlock <= blockNumber) {
    return checkpoint.delegatedTokenIds;
}
```

5. Результат вызова внешней функции следует кэшировать, а не вызывать функцию повторно (так как вызов внешних функций дорогой):
```solidity

// non optimized
uint256 tokenToEmit = (dailyEmission * (rewardToken.totalSupply() - rewardToken.balanceOf(address(ve)))) / rewardToken.totalSupply();
uint256 stakerReward = (tokenToEmit * rewardToken.balanceOf(address(ve))) / rewardToken.totalSupply();

// optimized
uint256 _totalSupply = rewardToken.totalSupply();
uint256 _balanceOf = rewardToken.balanceOf(address(ve));
uint256 tokenToEmit = (dailyEmission * (_totalSupply - _balanceOf)) / _totalSupply;
uint256 stakerReward = (tokenToEmit * _balanceOf) / _totalSupply;
```

6. Перед вызовом перевода необходимо проверить сумму на 0.

### Оптимизация смарт-контрактов и газ

1. В solidity не нужно создавать лишние переменные и функции, так как деплой будет дороже.
2. Не нужно для value type state variables присваивать значение по умолчанию, если оно не отлично от 0, так как так и так будет значение 0.
3. Упаповка state variables: если несколько переменных можно поместить в один слот (256 бит), то их нужно объявлять рядом.
4. Меньшая размерность переменной не означает меньше газа. Если переменная меньше 1 слота, но будет занимать 1 слот, то такая операция более дорогая, так как ее нужно будет предварительно обрезать.
5. Если есть возможность задавать статические значения для переменных, то лучше делать так, нежели вычислять их динамически при деплое.
6. Без необходимости не использовать промежуточные переменные, так как работа с ними тоже стоит газа.
7. По возможности лучше использовать mapping, а не динамические массивы.
8. Динамические массивы дороже, чем статические массивы (push требудет газа).
9. Для массивов важно, какая размерность у value type переменной, так как он упаковывает значения.
10. Лучше не создавать много мелких функций
11. В require лучше использовать более короткие message, желательно, чтоб строка умещалась в 32 байта.
12. Меньше менять переменную состояния.
13. Лучше использовать сторонние библиотеки.
