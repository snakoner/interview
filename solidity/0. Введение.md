### Простейший смарт-контракт


```solidity
// SPDX-License-Identifier: GPL-3.0

// означает, что код написан для версий компилятора solidity [0.4.16, 0.9.0). 
// Означает, что для других версий код может скомпилироваться некорректно
pragma solidity >=0.4.16 <0.9.0;

contract SimpleStorage {
    uint storedData; // uint (alias для uint256)

    function set(uint x) public {
        storedData = x;
    }

    function get() public view returns (uint) {
        return storedData;
    }
};
```

В Solidity контракт - это набор кода кода (функций) и данных (переменные состояния). Данный набор хранится по определенному адресу в блокчейне Ethereum.

Сейчас изменить функции set и get может любой, в дальнейшем будет показано, как сделать так, чтобы только создатель контракта мог это делать.

### Пример контракта для коина

> address - тип данных (20 байт), который описывает адрес. Не разрешено производить с адресом никакие арифметические операции. Адресом может быть адрес контракта либо hash(public_key) для внешнего аккаунта (то есть аккаунта кошелька)

> msg.sender - адрес пользователя, который вызвал функцию

> Ключевое слово public означает, что переменная состояния доступна извне, то есть ее значение может получить любой желающий. Для public переменных компилятор создает функцию с таким же названием, либо можно (но не нужно) создать данную функцию самому:


```solidity
function minter() external view returns (address) {
    return minter;
}
```


> Получить список ключей mapping нельзя, по умолчанию map[address] имеет значение 0. Для того, чтобы иметь список ключей, нужно использовать дополнительные типы данных для хранения.

> Для public mapping создается getter функция (она используется для получения баланса токена на определенном аккаунте):

```solidity
function balances(address account) external returns view (uint) {
    return balances[account];
}
```



```solidity
// SPDX-License-Identifier: GPL-3.0

// когда такое обозначение, это означает, что код не скомпилируется для компилятора версии ниже 0.8.26, но скомпилирует для версий от 0.8.26 до 0.9.0 не включительно
pragma solidity ^0.8.26;

contract Coin {
    // public делает переменную состояния доступной из вне
    address public minter;
    mapping (address => uint) public balances; // map[address] = balance

    // вызывается единственный раз при деплое контракта
    contstructor() {
        minter = msg.sender;
    }

    // Event в Solidity — это способ для контракта отправить сигнал во внешний мир при определённом действии или изменении состояния внутри контракта. Клиенты Ethereum, такие как веб-приложения, могут прослушивать эти события, передаваемые в блокчейн

    event Sent(address from, address to, uint amount);

    // Создатель контракта может создать новый токен в помощью функции mint (точнее присвоить весь supply на определенный адрес)
    function mint(address receiver, uint amount) public {
        require(msg.sender == minter); // если функцию вызывает не создатель, то error?
        balances[receiver] += amount;
    }

    // Ошибки позволяют предоставить информацию о том, почему операция невозможна. Они возвращаются аккаунту, вызвавшему функцию.
    error InsufficientBalance(uint requested, uint available);

    // Отправка токена от вызвавшего эту функцию на какой-то адрес
    function send(address receiver, uint amount) public {
        require(balances[msg.sender] >= amount, InsufficientBalance(amount, balances[msg.sender]));

        balances[msg.sender] -= amount;
        balances[receiver] += amount;
        emit Send(msg.sender, receiver, amount);
    }
};
```
