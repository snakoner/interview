Идея: взаимодействие с контрактом ведется не напрямую, а через посредника - прокси:

```
User --- tx ---- Proxy ---- | ----- Impl_V0
                            | ----- Impl_V1
                            | ----- Impl_V2
```

Работает через delegate call: логика лежит в Implementation, данные в Proxy.

Пример простейшей реализации:

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

contract Proxy {
    address public _impl;
    uint public _x;

    function setImplementation(address impl_) external {
        _impl = impl_;
    }

    function _delegate() private {
        (bool success, ) = _impl.delegatecall(msg.data);
        require(success, "failed");
    }

    // при передаче msg.data кода селектора, вызовется fallback -> _delegate -> delegatecall с нужным селектором
    fallback() external payable {
        _delegate();
    }

    receive() external payable { }
}

contract V1 {
    address public _impl;
    uint public _x;

    function inc() external {
        _x++;
    }

    function getIncSelector() external pure returns (bytes4) {
        return bytes4(abi.encodeWithSelector(this.inc.selector));
    }
}

contract V2 {
    address public _impl;
    uint public _x;

    function inc() external {
        _x += 10;
    }

    function getIncSelector() external pure returns (bytes4) {
        return bytes4(abi.encodeWithSelector(this.inc.selector));
    }
}
```
