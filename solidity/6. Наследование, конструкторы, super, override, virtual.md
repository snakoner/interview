### Наследование, множественное наследование

```solidity
contract Ownable {
  address pulbic owner;

  constructor() {
    owner = msg.sender;
  }

  modifier onlyOwner() {
    require(owner == msg.sender, "not an owner!");
    _;
  }
}

contract Balances is Ownable {
  function getBalance() external view onlyOwner returns(uint) {
    return address(this).balance;
  }

  function withdraw(address payable _to) external onlyOwner {
    _to.transfer(address(this).balance);
  }
}

contract MyContract is Ownable, Balances {
}

// or

contract MyContract is Balances {
}

```

Порядок следования при множественном наследовании важен. Сначала должен стоять самый высокоуровневый контракт (наиболее родительский: Ownable -> Balances).

Конструкторы вызываются в порядке от родителя к ребенку.

### Вызов родительских конструкторов

Абстрактный контракт - контракт, который нельзя развернуть **напрямую**, но от него можно наследоваться. Контракт нужно помечать абстрактным, когда хотя бы одна из его функций не реализована или когда они не предоставляют аргументов для конструкторов всех родительских классов.

Контракт **может быть** помечен как абстрактный, если мы не собираемся деплоить контракт напрямую.

В каком то плане абстрактные контракты похожи на интерфейсы, но в интерфейсах могуть быть только описания функций.

```solidity
contract Ownable {
  address pulbic owner;

  constructor(ownerOverride) {
    owner = ownerOverride == address(0) ? msg.sender : ownerOverride;
  }

  modifier onlyOwner() {
    require(owner == msg.sender, "not an owner!");
    _;
  }
}

abstract contract Balances is Ownable {
  function getBalance() external view onlyOwner returns(uint) {
    return address(this).balance;
  }

  function withdraw(address payable _to) external onlyOwner {
    _to.transfer(address(this).balance);
  }
}

// так можно делать, когда значение известно наперед
contract MyContract is Ownable(0x5B38Da6a701c568545dCfcB03FcB875f56beddC4), Balances {

}

// когда значение не известно
contract MyContract is Ownable, Balances {
  constructor(address _owner) Ownable(_owner) {

  }
}

```

### virtual, override

- `virtual` позволяет переопределить функцию в контракте потомке
- `override` говорит о том, что функция переопределена
- `virtual override` говорит о том, что функция переопределена и ее можно переопределить еще раз в контракте-потомке

```solidity
contract Ownable {
  address pulbic owner;

  constructor(ownerOverride) {
    owner = ownerOverride == address(0) ? msg.sender : ownerOverride;
  }

  modifier onlyOwner() {
    require(owner == msg.sender, "not an owner!");
    _;
  }

  function withdraw(address payable _to) external virtual onlyOwner {
    _to.transfer(address(this).balance);
  }
}

abstract contract Balances is Ownable {
  function getBalance() external view onlyOwner returns(uint) {
    return address(this).balance;
  }

  function withdraw(address payable _to) external virtual override onlyOwner {
    _to.transfer(getBalance());
  }
}

// 
contract MyContract is Ownable, Balances {
  function withdraw(address payable _to) external virtual override onlyOwner {
    // 1 вариант
    Balances.withdraw(_to);
    // 2 вариант
    super.withdraw(_to); // вызываем из контракта на одну иерархию вверх (Balances)
}
}
```

Так как мы наследуемся от двух родителей, то в MyContract нужно решить, какую из функций вызывать. Для этого надо переопределить функцию.
