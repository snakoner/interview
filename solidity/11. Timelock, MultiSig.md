### Timelock

Механизм, который добавляет транзакцию в очередь, но выполненить ее будет возможно только через заданное время. Также добавим возможность подтверждения несколькими аккаунтами этой транзакции.

```solidity

contract Timelock {
    address[] public owners;
    uint constant MIN_DELAY = 10;
    uint constant MAX_DELAY = 1000;
    uint constant GRACE_PERIOD = 10000;
    uint public constant CONFIRMATIONS_REQUIRED = 2;

    mapping(address => bool) public isOwner;
    mapping(bytes32=>bool) public queue;
    mapping(bytes32 => mapping(address => bool)) public confirmations;

    struct Transaction {
        bytes32 uid;    // hash
        address to;
        string func;
        bytes data;
        uint value;
        uint timestamp;
        uint confirmations; // сколько человек подтвердили эту транзакцию
    }

    mapping(bytes32 => Transaction) public txs;


    modifier onlyOwner() {
        require(isOwner[msg.sender], "not an owner");
        _;
    }

    event Queued(bytes32 id);
    event Discarded(bytes32 id);


    constructor(address[] memory _owners) {
        require(_owners.length >= CONFIRMATIONS_REQUIRED, "not enough owners");

        for (uint i = 0; i < _owners.length; i++) {
            require(_owners[i] == address(0), "owner is zero address");
            require(!isOwner[_owners[i]], "already in owners");

            isOwner[_owners[i]] = true;
            owners.push(_owners[i]);
        }
    }

    function addToQueue(
        address _to,
        string calldata _func,
        bytes calldata _data,
        uint _value,
        uint _timestamp
    ) external onlyOwner returns(bytes32) {
        require(_timestamp > block.timestamp + MIN_DELAY, "bad time");
        require(_timestamp < block.timestamp + MAX_DELAY, "bad time");
        bytes32 txId = keccak256(
            abi.encode(
                _to, 
                _func,
                _data,
                _value,
                _timestamp
        ));

        require(!queue[txId], "already in queue");

        queue[txId] = true;
        txs[txId] = Transaction({
            uid: txId,
            to: _to,
            func: _func,
            data: _data,
            value: _value,
            timestamp: _timestamp,
            confirmations: 0}
        );


        emit Queued(txId);

        return txId;
    }

    function confirm(bytes32 id) external onlyOwner {
        require(queue[id], "not queued");
        require(!confirmations[id][msg.sender], "confirmation already done");
        Transaction storage transaction = txs[id];
        transaction.confirmations++;
        confirmations[id][msg.sender] = true;
    }

    function cancelConfirmation(bytes32 id) external onlyOwner {
        require(queue[id], "not in queue");
        require(!confirmations[id][msg.sender], "not confirmed yet");
        confirmations[id][msg.sender] = false;
        Transaction storage transaction = txs[id];
        transaction.confirmations--;
     }

    function discart(bytes32 txId) external onlyOwner {
        require(queue[txId], "not queued");

        delete queue[txId];
        emit Discarded(txId);
    }

    function execute(      
        address _to,
        string calldata _func,
        bytes calldata _data,
        uint _value,
        uint _timestamp
    ) external payable onlyOwner returns(bytes memory) {
        require(_timestamp > block.timestamp, "too early");
        require(_timestamp < block.timestamp + GRACE_PERIOD, "too late");
        bytes32 txId = keccak256(
            abi.encode(
                _to, 
                _func,
                _data,
                _value,
                _timestamp
        ));

        bytes memory data;
        if (bytes(_func).length > 0) {
            data = abi.encodePacked(
                bytes4(keccak256(bytes(_func))), 
                data
            );            
        } else {
            data = _data;
        }

        require(queue[txId], "unknown tx");
        Transaction storage transaction = txs[txId];
        require(transaction.confirmations >= CONFIRMATIONS_REQUIRED, "not enough confirmations");

        (bool success, bytes memory response) = payable(_to).call{value:_value}(data);
        require(success);

        queue[txId] = false;

        return response;
    }
}

```
