```
unsafe.Pointer — это специальный тип в Go из пакета unsafe, который позволяет обходить систему типов и работать с памятью почти так же, как в C.

Обычные указатели в Go типобезопасны:
var x int = 42
var p *int = &x

var f *float64 = (*float64)(p) // ошибка компиляции

Но через unsafe:

import "unsafe"

var x int64 = 42

p := unsafe.Pointer(&x)
f := (*float64)(p)

fmt.Println(*f)
```

### Структура:

```
type slice struct {
    array unsafe.Pointer
    len   int
    cap   int
}
```

### Пуш в slice:

```
Аллокация нового куска памяти под новый массив в 2 раза больше при < 256 элем, и на 25% если больше 256 элем. (на самом деле там побитовый сдвиг уменьшающийся до 1.25х после 256) 

1. Копия старого массива на новое место. 
2. Перезапись указателя на новое место. 
3. Старое затирается в последствии GC.
```

### Срез

Использование slice[n:m] - создание нового слайса, который указывает на тот-же самый массив в памяти, где: 
1. len = m - n
2. cap = cap(old) - n

Пример:

```go
x := []int{1,2,3,4,5}
slice1 := x[:2] // [1, 2], len = 2, cap = 5
slice2 := x[1:4] // [2,3,4], len = 3, cap = 4
```


### Append
```go
func main() {
	a := make([]int, 0, 2)
	a = append(a, 1)
	_ = append(a, 2)
	fmt.Println(a)          // [1], так как len увеличилась в копии, но мы не присваиваем результат append
	fmt.Println(a[:cap(a)]) // [1, 2]
}
```

### Задачи на slice
1.

```
func main() {
    s := []int{1, 2, 3}
    t := append(s, 4)
    fmt.Println(s) // 1, 2, 3
    fmt.Println(t) // 1, 2, 3, 4 (копия)
}
```

2.

```go
func main() {
    s := make([]int, 3, 5)
    s[0] = 1
    s[1] = 2
    s[2] = 3

    t := append(s, 4)
    fmt.Println(s) // [1, 2, 3] ptr = 0xA
    fmt.Println(t) // [1, 2, 3, 4] ptr = 0xA
}
```

3.
```go
func main() {
    x := []int{1, 2, 3, 4, 5}
    a := x[:2] // [1, 2], len = 2, cap = 5

    b := append(a, 10) // b = [1, 2, 10], len = 3, cap = 5 || a = [1, 2], len = 2, cap = 5

    fmt.Println(x) // x = [1, 2, 10, 4, 5]
    fmt.Println(a) // a = [1, 2]
    fmt.Println(b) // b = [1, 2, 10]
}
```

4.

```go
func main() {
    s := make([]int, 0, 3)

    append(s, 1) // backing = [1]
    append(s, 2) // backing = [2]

    fmt.Println(s) // [], так как len в s не изменилась
}
```

5.

```
func main() {
    s := make([]int, 1, 2) // [0], len = 1, cap = 2

    s[0] = 1 // [1], len = 1, cap = 2

    t := append(s, 2) // t = [1, 2], len = 2, cap = 2 || s = [1], len = 1, cap = 2
    u := append(s, 3) // u = [1, 3], len = 2, cap = 2 || s = [1], len = 1, cap = 2

    fmt.Println(t) // t = [1, 3]
    fmt.Println(u) // u = [1, 3]
}
```
