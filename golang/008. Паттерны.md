### Options
Удобная настройка объекта без огромного конструктора.

```
type Server struct {
	host    string
	port    int
	timeout time.Duration
}

type Option func(*Server)

func WithHost(host string) Option {
	return func(s *Server) {
		s.host = host
	}
}

func WithPort(port int) Option {
	return func(s *Server) {
		s.port = port
	}
}

func WithTimeout(timeout time.Duration) Option {
	return func(s *Server) {
		s.timeout = timeout
	}
}

func NewServer(opts ...Option) *Server {
	s := &Server{
		host:    "127.0.0.1",
		port:    80,
		timeout: 3 * time.Second,
	}

	for _, opt := range opts {
		opt(s)
	}

	return s
}
```

### Semaphore
Semaphore ограничивает количество одновременно выполняемых операций.
```
func main() {
	sem := make(chan struct{}, 3)
	var wg sync.WaitGroup

	for i := 1; i <= 10; i++ {
		wg.Add(1)

		go func(id int) {
			defer wg.Done()

			sem <- struct{}{}
			defer func() { <-sem }()

			fmt.Println("start", id)
			time.Sleep(time.Second)
			fmt.Println("done", id)
		}(i)
	}

	wg.Wait()
}
```
