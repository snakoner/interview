Изящное завершение(Grateful shutdown) работы заключается в том, что программа завершается после:
1. Завершения всех незавершенных процессов (веб-запрос, циклы) - не должно запускаться никаких новых процессов и не должно приниматься никаких новых веб-запросов.
2. Закрытия всех открытых соединений с внешними сервисами и базами данных.


```go
func f(ctx context.Context, number int) {
	for {
		select {
		case <-ctx.Done():
			fmt.Println("im done")
			return
		case <-time.After(time.Second):
			fmt.Println("f im working", number)
		}
	}
}

func main() {
	ctx, cancel := context.WithCancel(context.Background())
	wg := sync.WaitGroup{}

	go func() {
		exit := make(chan os.Signal, 1)
		signal.Notify(exit, os.Interrupt, os.Kill)

		<-exit
		cancel()
	}()

	wg.Add(1)
	go func() {
		f(ctx, 1)
		wg.Done()
	}()

	wg.Add(1)
	go func() {
		f(ctx, 2)
		wg.Done()
	}()

	wg.Wait()
}

```
